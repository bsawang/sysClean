# SysClean Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Claude Code Skill that provides a Web-based C: drive cleanup tool with category scanning, real-time progress via SSE, risk-labeled results, and confirmed file deletion.

**Architecture:** Python Flask backend serving a single HTMX + SSE frontend page. Scanning uses `os.walk` with generators yielding results to an SSE event stream. Risk levels determined by path-pattern matching. Cleanup uses `shutil.rmtree` / `os.remove` with file-lock checks and logging.

**Tech Stack:** Python 3.9, Flask 3.x, HTMX 2.x (CDN), SSE, psutil, Lucide Icons (inline SVG), i18n (JSON language packs)

```python
import fnmatch
import os

# Risk rules: (glob_pattern, risk_level, description)
RISK_RULES = [

    (r"*\Windows\Temp\*",                 RiskLevel.SAFE,    "Windows 临时文件，可安全删除"),
    (r"*\AppData\Local\Temp\*",           RiskLevel.SAFE,    "用户临时文件，重启后可安全删除"),

    (r"*\.vscode\extensions\*",           RiskLevel.CAUTION, "VS Code 扩展，删除后需重新安装"),
    (r"*\JianyingPro\User Data\*",        RiskLevel.CAUTION, "剪映用户数据，包含模板和模型文件"),
    (r"*\JianyingPro\Apps\*",            RiskLevel.CAUTION, "剪映应用文件，删除后需重新安装"),
    (r"*\StreetFighterV\*",              RiskLevel.CAUTION, "游戏数据文件，删除后需重新下载"),
    (r"*\Program Files\WindowsApps\*",    RiskLevel.CAUTION, "Windows 应用商店应用，删除后可能影响功能"),
    # --- DANGER (rare, only for explicit system paths) ---
    (r"*\Windows\System32\*",             RiskLevel.DANGER,  "系统核心目录，禁止操作"),
    (r"*\Windows\WinSxS\*",              RiskLevel.DANGER,  "系统组件存储，删除会破坏系统"),
]


def classify_risk(path: str, size: int = 0) -> tuple[RiskLevel, str]:
    """Match a path against risk rules. Returns (risk_level, description)."""
    for pattern, level, desc in RISK_RULES:
        if fnmatch.fnmatch(path, pattern):
            return level, desc
    # Default: large files get CAUTION, small unknowns get SAFE
    if size > 500 * 1024 * 1024:  # > 500 MB
        return RiskLevel.CAUTION, "大文件，请确认是否需要"
    return RiskLevel.SAFE, "可安全删除"
```

```python
import os
import shutil
import glob as glob_module
from pathlib import Path


class ScannerEngine:
    """Scans disk categories and yields ScanItem generators."""

    def __init__(self):
        self.user_home = Path(os.environ.get("USERPROFILE", "C:\\Users\\default"))
        self.counter = 0

    def _next_id(self, prefix: str) -> str:
        self.counter += 1
        return f"{prefix}_{self.counter}"

    def scan_temp(self):
        """Scan Windows Temp and User Temp directories."""
        dirs = [
            ("C:\\Windows\\Temp", "Windows 系统临时文件目录"),
            (str(self.user_home / "AppData" / "Local" / "Temp"), "当前用户临时文件目录"),
        ]
        for base_path, desc in dirs:
            if not os.path.exists(base_path):
                continue
            total_size = 0
            file_count = 0
            for root, dirs_list, files in os.walk(base_path):
                for f in files:
                    try:
                        fp = os.path.join(root, f)
                        total_size += os.path.getsize(fp)
                        file_count += 1
                    except OSError:
                        pass
            risk, risk_desc = classify_risk(base_path)
            yield ScanItem(
                id=self._next_id("temp"),
                category="temp",
                path=base_path,
                size=total_size,
                risk=risk,
                risk_desc=risk_desc,
                item_count=file_count,
            )

    def scan_cache(self):
        """Scan NVIDIA DXCache, GLCache, and other GPU caches."""
        cache_dirs = [
            str(self.user_home / "AppData" / "Local" / "NVIDIA" / "DXCache"),
            str(self.user_home / "AppData" / "Local" / "NVIDIA" / "GLCache"),
        ]
        for base_path in cache_dirs:
            if not os.path.exists(base_path):
                continue
            total_size = 0
            file_count = 0
            for root, dirs_list, files in os.walk(base_path):
                for f in files:
                    try:
                        fp = os.path.join(root, f)
                        total_size += os.path.getsize(fp)
                        file_count += 1
                    except OSError:
                        pass
            risk, risk_desc = classify_risk(base_path)
            if file_count > 0:
                yield ScanItem(
                    id=self._next_id("cache"),
                    category="cache",
                    path=base_path,
                    size=total_size,
                    risk=risk,
                    risk_desc=risk_desc,
                    item_count=file_count,
                )

    def scan_installer(self):
        """Scan *-updater\installer.exe and similar installer residue."""
        # Common updater directories
        updater_patterns = [
            str(self.user_home / "AppData" / "Local" / "*updater" / "installer.exe"),
            str(self.user_home / "AppData" / "Local" / "*updater" / "pending" / "*.exe"),
            str(self.user_home / "AppData" / "Local" / "app_shell_cache*" / "*.exe"),
        ]
        seen = set()
        for pattern in updater_patterns:
            for fp in glob_module.glob(pattern, recursive=False):
                if fp in seen:
                    continue
                seen.add(fp)
                try:
                    size = os.path.getsize(fp)
                except OSError:
                    continue
                risk, risk_desc = classify_risk(fp)
                yield ScanItem(
                    id=self._next_id("inst"),
                    category="installer",
                    path=fp,
                    size=size,
                    risk=risk,
                    risk_desc=risk_desc,
                )

    def scan_browser(self):
        """Scan browser cache directories (Chrome)."""
        chrome_cache = (
            self.user_home / "AppData" / "Local" / "Google" / "Chrome" / "User Data" / "Default" / "Cache"
        )
        chrome_code_cache = (
            self.user_home / "AppData" / "Local" / "Google" / "Chrome" / "User Data" / "Default" / "Code Cache"
        )
        for base_path in [chrome_cache, chrome_code_cache]:
            if not os.path.exists(base_path):
                continue
            total_size = 0
            file_count = 0
            for root, dirs_list, files in os.walk(base_path):
                for f in files:
                    try:
                        fp = os.path.join(root, f)
                        total_size += os.path.getsize(fp)
                        file_count += 1
                    except OSError:
                        pass
            if file_count > 0:
                risk, risk_desc = classify_risk(str(base_path))
                yield ScanItem(
                    id=self._next_id("browser"),
                    category="browser",
                    path=str(base_path),
                    size=total_size,
                    risk=risk,
                    risk_desc=risk_desc,
                    item_count=file_count,
                )

    def scan(self, categories: list[str]) -> list[ScanItem]:
        """Run selected categories and return all results."""
        dispatcher = {
            "temp": self.scan_temp,
            "cache": self.scan_cache,
            "installer": self.scan_installer,
            "browser": self.scan_browser,
        }
        results = []
        for cat in categories:
            if cat in dispatcher:
                results.extend(list(dispatcher[cat]()))
        return results
```

```python
import os
import shutil
import logging
import time
from datetime import datetime
from pathlib import Path


class CleanExecutor:
    """Executes file deletion with safety checks and progress tracking."""

    def __init__(self, log_dir: str = "logs"):
        self.log_dir = Path(log_dir)
        self.log_dir.mkdir(parents=True, exist_ok=True)
        self._setup_logging()
        self.total_items = 0
        self.completed = 0
        self.failed = []
        self.total_freed = 0

    def _setup_logging(self):
        log_file = self.log_dir / f"sysclean-{datetime.now().strftime('%Y-%m-%d')}.log"
        self.logger = logging.getLogger("sysclean")
        self.logger.setLevel(logging.INFO)
        # File handler
        fh = logging.FileHandler(str(log_file), encoding="utf-8")
        fh.setFormatter(logging.Formatter("%(asctime)s [%(levelname)s] %(message)s"))
        self.logger.addHandler(fh)

    def _is_file_locked(self, path: str) -> bool:
        """Check if a file is in use by another process."""
        try:
            if os.path.isfile(path):
                with open(path, "r+b") as f:
                    f.read(0)
            elif os.path.isdir(path):
                # Just check if we can list the directory
                os.listdir(path)
            return False
        except (PermissionError, OSError):
            return True

    def delete_item(self, item) -> bool:
        """Delete a single ScanItem. Returns True on success."""
        path = item.path
        if not os.path.exists(path):
            self.logger.warning(f"路径不存在，跳过: {path}")
            return True  # Already gone = success

        if self._is_file_locked(path):
            self.logger.error(f"文件被占用，跳过: {path}")
            return False

        try:
            if os.path.isfile(path):
                size = os.path.getsize(path)
                os.remove(path)
                freed = size
            elif os.path.isdir(path):
                size = 0
                for root, dirs, files in os.walk(path):
                    for f in files:
                        try:
                            size += os.path.getsize(os.path.join(root, f))
                        except OSError:
                            pass
                shutil.rmtree(path, ignore_errors=True)
                freed = size
            else:
                return False

            self.total_freed += freed
            self.logger.info(
                f"CLEAN OK | {path} | {self._format_size(freed)} | "
                f"risk={item.risk.value} | cat={item.category}"
            )
            return True

        except Exception as e:
            self.logger.error(f"CLEAN FAIL | {path} | {e}")
            return False

    def clean_items(self, items: list) -> dict:
        """Execute cleanup for a list of ScanItems. Returns summary dict."""
        self.total_items = len(items)
        self.completed = 0
        self.failed = []
        self.total_freed = 0

        for item in items:
            success = self.delete_item(item)
            if success:
                self.completed += 1
            else:
                self.failed.append(item.path)
            # Yield progress for SSE
            yield {
                "type": "clean_progress",
                "completed": self.completed,
                "total": self.total_items,
                "path": item.path,
                "success": success,
                "freed_so_far": self.total_freed,
            }

        # Final summary
        yield {
            "type": "clean_complete",
            "total_items": self.total_items,
            "completed": self.completed,
            "failed_count": len(self.failed),
            "total_freed": self.total_freed,
        }

    @staticmethod
    def _format_size(size: int) -> str:
        for unit in ("B", "KB", "MB", "GB", "TB"):
            if size < 1024:
                return f"{size:.1f} {unit}"
            size /= 1024
        return f"{size:.1f} PB"
```

```python
import json
import threading
from flask import Flask, render_template, request, Response, jsonify, g

from scanner import ScannerEngine, ScanItem
from cleaner import CleanExecutor
from i18n import get_text, get_all_texts

app = Flask(__name__)

# In-memory state
scan_results: list[ScanItem] = []
scan_engine = ScannerEngine()


@app.before_request
def set_language():
    """Detect language from query param ?lang=en|zh, default zh."""
    g.lang = request.args.get("lang", "zh")
    if g.lang not in ("zh", "en"):
        g.lang = "zh"


def _(key: str, **kwargs) -> str:
    """Shortcut for get_text in templates."""
    return get_text(key, getattr(g, "lang", "zh"), **kwargs)


@app.route("/")
def index():
    return render_template("index.html", lang=getattr(g, "lang", "zh"))


@app.route("/api/translations")
def api_translations():
    """Return language pack for the frontend JS."""
    return jsonify(get_all_texts(getattr(g, "lang", "zh")))


@app.route("/api/disk")
def api_disk():
    """Return C: drive disk usage info."""
    import psutil
    usage = psutil.disk_usage("C:/")
    return jsonify({
        "total": usage.total,
        "used": usage.used,
        "free": usage.free,
        "percent": usage.percent,
    })


@app.route("/api/scan/start", methods=["POST"])
def api_scan_start():
    """Start scanning. Body: {"categories": ["temp", "cache", ...]}"""
    global scan_results
    data = request.get_json()
    categories = data.get("categories", [])
    if not categories:
        return jsonify({"error": "请至少选择一个分类"}), 400

    # Run scan in background thread, store results
    def do_scan():
        global scan_results
        scan_results = scan_engine.scan(categories)

    thread = threading.Thread(target=do_scan, daemon=True)
    thread.start()
    return jsonify({"status": "started", "categories": categories})


@app.route("/api/scan/results")
def api_scan_results():
    """Return scan results as JSON."""
    global scan_results
    items = []
    for item in scan_results:
        items.append({
            "id": item.id,
            "category": item.category,
            "path": item.path,
            "size": item.size,
            "size_fmt": CleanExecutor._format_size(item.size),
            "risk": item.risk.value,
            "risk_label": {"safe": "安全", "caution": "注意", "danger": "高风险"}[item.risk.value],
            "risk_desc": item.risk_desc,
            "item_count": item.item_count,
        })
    return jsonify({"items": items, "total": len(items)})


@app.route("/api/clean/start", methods=["POST"])
def api_clean_start():
    """Start cleaning selected items. Body: {"ids": ["id1", "id2", ...]}"""
    global scan_results
    data = request.get_json()
    ids = set(data.get("ids", []))

    selected = [item for item in scan_results if item.id in ids]
    if not selected:
        return jsonify({"error": "没有选中任何清理项"}), 400

    executor = CleanExecutor()

    def progress_generator():
        for event in executor.clean_items(selected):
            yield f"data: {json.dumps(event)}\n\n"

    return Response(progress_generator(), mimetype="text/event-stream")


@app.route("/api/processes")
def api_processes():
    """Return top disk/memory consuming processes."""
    import psutil
    processes = []
    for proc in psutil.process_iter(["pid", "name", "memory_info", "io_counters"]):
        try:
            info = proc.info
            processes.append({
                "pid": info["pid"],
                "name": info["name"],
                "memory_mb": round((info["memory_info"].rss if info["memory_info"] else 0) / 1048576, 1),
            })
        except (psutil.NoSuchProcess, psutil.AccessDenied):
            pass

    processes.sort(key=lambda p: -p["memory_mb"])
    return jsonify(processes[:30])


if __name__ == "__main__":
    print("🧹 SysClean 启动中...")
    print("   访问 http://localhost:5000 打开清理工具")
    app.run(host="127.0.0.1", port=5000, debug=True)
```

```python
import json
import os
from pathlib import Path

_I18N_DIR = Path(__file__).parent

# Cache loaded language packs
_cache: dict[str, dict[str, str]] = {}


def _load_lang(lang: str) -> dict[str, str]:
    """Load a language pack JSON file."""
    if lang in _cache:
        return _cache[lang]
    file_path = _I18N_DIR / f"{lang}.json"
    if not file_path.exists():
        file_path = _I18N_DIR / "zh.json"  # fallback to Chinese
    with open(file_path, "r", encoding="utf-8") as f:
        _cache[lang] = json.load(f)
    return _cache[lang]


def get_text(key: str, lang: str = "zh", **kwargs) -> str:
    """Get translated text by key. Falls back to key name if not found."""
    pack = _load_lang(lang)
    text = pack.get(key, key)
    if kwargs:
        text = text.format(**kwargs)
    return text


def get_all_texts(lang: str = "zh") -> dict:
    """Return the entire language pack as a dict (for embedding in JS)."""
    return _load_lang(lang)
```

```html
// ===== i18n =====
let _T = {};  // translations loaded from server

async function loadTranslations(lang) {
  try {
    const resp = await fetch(`/api/translations?lang=${lang}`);
    _T = await resp.json();
    document.querySelector('html').setAttribute('lang', lang);
    // Update all DOM elements with data-i18n keys
    document.querySelectorAll('[data-i18n]').forEach(el => {
      el.textContent = _t(el.getAttribute('data-i18n'));
    });
    document.querySelectorAll('[data-i18n-placeholder]').forEach(el => {
      el.setAttribute('placeholder', _t(el.getAttribute('data-i18n-placeholder')));
    });
  } catch(e) { console.warn('i18n load failed', e); }
}

function _t(key) { return _T[key] || key; }

function switchLang(lang) {
  const url = new URL(window.location);

// ===== Helper: Format Bytes =====
function fmtSize(bytes) {
  if (bytes === 0) return '0 B';
  const units = ['B', 'KB', 'MB', 'GB', 'TB'];
  let i = 0;
  let size = bytes;
  while (size >= 1024 && i < units.length - 1) { size /= 1024; i++; }
  return size.toFixed(i > 0 ? 1 : 0) + ' ' + units[i];
}

// ===== Helper: Risk Tag HTML =====
function riskTag(risk) {
  const map = {safe: ['risk-safe', '安全'], caution: ['risk-caution', '注意'], danger: ['risk-danger', '高风险']};
  const [cls, label] = map[risk] || map.safe;
  return `<span class="risk-tag ${cls}">${label}</span>`;
}

    arrow.style.transform = 'rotate(-90deg)';
  }
}

// ===== Get Selected Items =====
function getSelectedItems() {
  // Parse from the rendered results
  const items = [];
  const groups = document.querySelectorAll('.result-group');
  groups.forEach(group => {
    const headerCheck = group.querySelector('.result-header > input[type="checkbox"]');
    const isGroupChecked = headerCheck ? headerCheck.checked : false;
    const catEl = group.querySelector('.result-header span[style*="flex:1"]');
    const catName = catEl ? catEl.textContent.trim() : '';
    const catMap = {'临时文件':'temp','系统缓存':'cache','安装程序残留':'installer','浏览器缓存':'browser'};
    const category = catMap[catName] || catName;

    const itemRows = group.querySelectorAll('.result-item');
    itemRows.forEach(row => {
      const cb = row.querySelector('input[type="checkbox"]');
      if (cb && (cb.checked || isGroupChecked)) {
        // Parse size from the size span
        const sizeSpan = row.querySelector('span[style*="white-space"]') || row.querySelector('span:last-child');
        const pathSpan = row.querySelector('.result-path');
        let sizeBytes = 0;
        if (sizeSpan) {
          const txt = sizeSpan.textContent.trim();
          const match = txt.match(/^([\d.]+)\s*(B|KB|MB|GB|TB)$/);
          if (match) {
            const val = parseFloat(match[1]);
            const unit = match[2];
            const mult = {B:1, KB:1024, MB:1048576, GB:1073741824, TB:1099511627776};
            sizeBytes = val * (mult[unit] || 1);
          }
        }
        // Get risk from the risk tag
        const riskTagEl = row.querySelector('.risk-tag');
        let risk = 'safe';
        if (riskTagEl) {
          if (riskTagEl.classList.contains('risk-safe')) risk = 'safe';
          else if (riskTagEl.classList.contains('risk-caution')) risk = 'caution';
          else if (riskTagEl.classList.contains('risk-danger')) risk = 'danger';
        }
        items.push({
          id: 'selected_' + items.length,
          category: category,
          path: pathSpan ? pathSpan.textContent.trim() : '',
          size: sizeBytes,
          risk: risk,
          risk_desc: ''
        });
      }
    });
  });
  return items;
}

      </div>
    </div>

    const resp = await fetch('/api/clean/start', {
      method: 'POST',
      headers: {'Content-Type': 'application/json'},
      body: JSON.stringify({ids: selected.map((_, i) => 'selected_' + i)})
    });

    const reader = resp.body.getReader();
    const decoder = new TextDecoder();
    let buffer = '';

    while (true) {
      const {done, value} = await reader.read();
      if (done) break;
      buffer += decoder.decode(value, {stream: true});
      const lines = buffer.split('\n\n');
      buffer = lines.pop() || '';

      for (const line of lines) {
        if (!line.startsWith('data: ')) continue;
        const data = JSON.parse(line.slice(6));

        if (data.type === 'clean_progress') {
          const pct = Math.round((data.completed / data.total) * 100);
          document.getElementById('clean-pct').textContent = pct + '%';
          document.getElementById('clean-bar').style.width = pct + '%';
          document.getElementById('clean-done').textContent = `${data.completed}/${data.total}`;
          document.getElementById('clean-freed-val').textContent = fmtSize(data.freed_so_far);

'<svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="#3fb950" stroke-width="2"><circle cx="12" cy="12" r="10"/><polyline points="8 12 11 15 16 10"/></svg>' : '<svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="#f85149" stroke-width="2"><circle cx="12" cy="12" r="10"/><line x1="15" y1="9" x2="9" y2="15"/><line x1="9" y1="9" x2="15" y2="15"/></svg>';
          const status = data.success ? '<span class="text-success" style="font-size:0.8em;">已完成</span>' : '<span class="text-danger" style="font-size:0.8em;">失败</span>';
          const feed = document.getElementById('clean-feed');
          feed.insertAdjacentHTML('beforeend', `<div class="feed-row">${icon}<span class="mono" style="flex:1;color:#8b949e;">${data.path}</span>${status}</div>`);
        }

        if (data.type === 'clean_complete') {
          // Show complete report
          document.getElementById('clean-pct').textContent = '100%';
          document.getElementById('clean-bar').style.width = '100%';
          setTimeout(() => showComplete(data), 500);
        }
      }
    }
  } catch(e) {
    alert('清理失败: ' + e.message);
  }
}

    if (pctEl) {
      const pct = parseFloat(pctEl.textContent);
      if (!isNaN(pct)) {
        const el = document.getElementById('disk-info');

          pctEl.className = 'disk-percent ' + (pct > 90 ? 'danger' : pct > 75 ? 'warning' : 'ok');
        }
      }
    }
  }
});
```
